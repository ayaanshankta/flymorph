# FlyMorph Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A TypeScript monorepo that measures Drosophila larva area (classical CV) and adult head/thorax/abdomen area (SAM-distilled U-Net with MC-dropout uncertainty) in mm², in the browser and a Node CLI, plus an explainer website.

**Architecture:** `packages/core` holds every algorithm as pure TS functions over typed arrays (no DOM, no Node APIs). `apps/web` (Vite+React) and `apps/cli` (Node) only decode images and run the ONNX model, then call core. `ml/` (Python) drafts labels with SAM, serves them to the review page, trains the student, exports ONNX.

**Tech Stack:** TypeScript 5, npm workspaces, Vitest, Vite, React 18, onnxruntime-web / onnxruntime-node, sharp; Python 3.12, PyTorch 2.7 (MPS), torchvision 0.22, transformers (SAM ViT-B), Flask, onnx.

**Spec:** `docs/spec.md`

## Global Constraints

- Originals in `~/Downloads/FlyAreaDetect/` are read-only; never copy the tree (16 GB free disk). Cache only downscaled images (long side 1024) under `ml/data/` (git-ignored).
- Commits as `AyaanShankta <ayaanshankta08@gmail.com>`, no co-author trailer. Repo private.
- Mask label values: 0 background, 1 head, 2 thorax, 3 abdomen.
- Scale: micrometer bar = 1 mm, 100 ticks; `pxPerMm = 100 × tickPeriodPx`.
- Model I/O: `image` float32 [B,3,512,512] ImageNet-normalised; `drop` float32 [B,576,1,1]; output `probs` float32 [B,4,512,512] (softmax).
- MC passes T = 8, drop p = 0.2; flag a fly if sd/mean > 0.10 for body area.
- Accuracy only on approved masks from held-out sessions; Dice > 0.98 is treated as a leak until disproven.
- Each milestone ends with `git tag mNN-<name>`.

## Review Focus

1. Wrong/blank image given as the scale → calibration must return low confidence (< 5), not a confident garbage px/mm; UI then asks for 2 clicks. (Task 4 test)
2. Larva cut off by the image border → still measured but flagged `edge`. (Task 3 test)
3. Two larvae touching → one blob of ~double area → flagged `merged?` when area > 1.8× median. (Task 3 test)
4. Image with no larvae / no fly → empty result, no crash, UI says "none found". (Task 3 + Task 9 tests)
5. Grayscale or alpha-channel inputs → decode to RGBA; `toValue` works on any RGBA. (Task 2 test)

```ts
export type RGB = { w: number; h: number; data: Uint8ClampedArray };
export type Gray = { w: number; h: number; data: Uint8Array };

export const gray = (w: number, h: number): Gray => ({ w, h, data: new Uint8Array(w * h) });

export function toValue(img: RGB): Gray {
  const out = gray(img.w, img.h);
  for (let i = 0, j = 0; j < out.data.length; i += 4, j++)
    out.data[j] = Math.max(img.data[i], img.data[i + 1], img.data[i + 2]);
  return out;
}

const K = [1, 4, 6, 4, 1];
// Separable: blur rows, then columns. Edges clamp to the nearest pixel.
export function gaussianBlur(g: Gray): Gray {
  const { w, h } = g, tmp = new Float32Array(w * h), out = gray(w, h);
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
    let s = 0;
    for (let k = -2; k <= 2; k++) s += K[k + 2] * g.data[y * w + Math.min(w - 1, Math.max(0, x + k))];
    tmp[y * w + x] = s / 16;
  }
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
    let s = 0;
    for (let k = -2; k <= 2; k++) s += K[k + 2] * tmp[Math.min(h - 1, Math.max(0, y + k)) * w + x];
    out.data[y * w + x] = Math.round(s / 16);
  }
  return out;
}

export function resizeBilinear(img: RGB, w: number, h: number): RGB {
  const out = new Uint8ClampedArray(w * h * 4), sx = img.w / w, sy = img.h / h;
  for (let y = 0; y < h; y++) {
    const fy = Math.min(img.h - 1, Math.max(0, (y + 0.5) * sy - 0.5)), y0 = Math.floor(fy),
      y1 = Math.min(img.h - 1, y0 + 1), ty = fy - y0;
    for (let x = 0; x < w; x++) {
      const fx = Math.min(img.w - 1, Math.max(0, (x + 0.5) * sx - 0.5)), x0 = Math.floor(fx),
        x1 = Math.min(img.w - 1, x0 + 1), tx = fx - x0;
      for (let c = 0; c < 4; c++) {
        const p = (yy: number, xx: number) => img.data[(yy * img.w + xx) * 4 + c];
        out[(y * w + x) * 4 + c] =
          (p(y0, x0) * (1 - tx) + p(y0, x1) * tx) * (1 - ty) + (p(y1, x0) * (1 - tx) + p(y1, x1) * tx) * ty;
      }
    }
  }
  return { w, h, data: out };
}
```

```tsx
import { useEffect, useRef } from 'react';
import type { RGB } from '@flymorph/core';
export type Box = { x: number; y: number; w: number; h: number; text: string };
type Props = { img: RGB; labels?: ArrayLike<number>; color?: (l: number) => [number, number, number] | null;
  boxes?: Box[]; onClick?: (x: number, y: number, e: React.MouseEvent) => void };
export function Overlay({ img, labels, color, boxes = [], onClick }: Props) {
  const ref = useRef<HTMLCanvasElement>(null);
  useEffect(() => {
    const c = ref.current!; c.width = img.w; c.height = img.h;
    const ctx = c.getContext('2d')!, px = new ImageData(new Uint8ClampedArray(img.data), img.w, img.h);
    if (labels && color) for (let i = 0; i < labels.length; i++) {
      const col = color(labels[i]); if (!col) continue;
      for (let k = 0; k < 3; k++) px.data[i * 4 + k] = px.data[i * 4 + k] * 0.5 + col[k] * 0.5;
    }
    ctx.putImageData(px, 0, 0);
    ctx.lineWidth = Math.max(2, img.w / 600); ctx.font = `${Math.max(14, img.w / 60)}px system-ui`;
    for (const b of boxes) {
      ctx.strokeStyle = ctx.fillStyle = '#39ff88';
      ctx.strokeRect(b.x, b.y, b.w, b.h); ctx.fillText(b.text, b.x, Math.max(20, b.y - 8));
    }
  }, [img, labels, color, boxes]);
  const click = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!onClick) return; const r = e.currentTarget.getBoundingClientRect();
    onClick(((e.clientX - r.left) / r.width) * img.w, ((e.clientY - r.top) / r.height) * img.h, e);
  };
  return <canvas ref={ref} onClick={click} className="overlay" />;
}
```

```ts
import type { Gray } from './image';
// Otsu: pick t maximising between-class variance of the histogram.
export function otsu(g: Gray): number {
  const hist = new Float64Array(256); for (const v of g.data) hist[v]++;
  const n = g.data.length; let sum = 0; for (let i = 0; i < 256; i++) sum += i * hist[i];
  let wB = 0, sumB = 0, best = 0, t = 0;
  for (let i = 0; i < 256; i++) {
    wB += hist[i]; if (!wB) continue; const wF = n - wB; if (!wF) break;
    sumB += i * hist[i]; const mB = sumB / wB, mF = (sum - sumB) / wF, between = wB * wF * (mB - mF) ** 2;
    if (between > best) { best = between; t = i; }
  }
  return t;
}
export function binarize(g: Gray, t: number): Gray {
  return { w: g.w, h: g.h, data: g.data.map((v) => (v > t ? 1 : 0)) };
}
```

```ts
import type { Gray } from './image';
// Square structuring element of size 2r+1, done separably (rows then columns): O(n·r) not O(n·r²).
function pass(m: Gray, r: number, keepIfAny: boolean): Gray {
  const { w, h } = m, tmp = new Uint8Array(w * h), out = new Uint8Array(w * h);
  const run = (get: (i: number) => number, set: (i: number, v: number) => void, len: number) => {
    for (let i = 0; i < len; i++) {
      let v = keepIfAny ? 0 : 1;
      for (let k = Math.max(0, i - r); k <= Math.min(len - 1, i + r); k++) {
        const p = get(k); if (keepIfAny ? p : !p) { v = keepIfAny ? 1 : 0; break; }
      }
      set(i, v);
    }
  };
  for (let y = 0; y < h; y++) run((k) => m.data[y * w + k], (i, v) => (tmp[y * w + i] = v), w);
  for (let x = 0; x < w; x++) run((k) => tmp[k * w + x], (i, v) => (out[i * w + x] = v), h);
  return { w, h, data: out };
}
export const dilate = (m: Gray, r: number) => pass(m, r, true);
export const erode = (m: Gray, r: number) => pass(m, r, false);
export const open = (m: Gray, r: number) => dilate(erode(m, r), r);   // removes specks
export const close = (m: Gray, r: number) => erode(dilate(m, r), r);  // bridges gaps
```

```ts
import type { Gray } from './image';
// Flood the background from every border pixel; background the flood can't reach is a hole.
export function fillHoles(m: Gray): Gray {
  const { w, h } = m, seen = new Uint8Array(w * h), stack: number[] = [];
  const push = (i: number) => { if (!m.data[i] && !seen[i]) { seen[i] = 1; stack.push(i); } };
  for (let x = 0; x < w; x++) { push(x); push((h - 1) * w + x); }
  for (let y = 0; y < h; y++) { push(y * w); push(y * w + w - 1); }
  while (stack.length) {
    const i = stack.pop()!, x = i % w, y = (i - x) / w;
    if (x > 0) push(i - 1); if (x < w - 1) push(i + 1); if (y > 0) push(i - w); if (y < h - 1) push(i + w);
  }
  return { w, h, data: m.data.map((v, i) => (v || !seen[i] ? 1 : 0)) };
}
```

```ts
export type Component = { id: number; area: number; x: number; y: number; w: number; h: number; cx: number; cy: number; edge: boolean };
// Flood-fill labelling (8-connected). Components come out in raster order of their first pixel.
export function components(m: Gray): { labels: Int32Array; comps: Component[] } {
  const { w, h } = m, labels = new Int32Array(w * h), comps: Component[] = [], stack: number[] = [];
  for (let s = 0; s < w * h; s++) {
    if (!m.data[s] || labels[s]) continue;
    const id = comps.length + 1; let area = 0, sx = 0, sy = 0, x0 = w, y0 = h, x1 = 0, y1 = 0;
    labels[s] = id; stack.push(s);
    while (stack.length) {
      const i = stack.pop()!, x = i % w, y = (i - x) / w;
      area++; sx += x; sy += y; x0 = Math.min(x0, x); y0 = Math.min(y0, y); x1 = Math.max(x1, x); y1 = Math.max(y1, y);
      for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) {
        const nx = x + dx, ny = y + dy, j = ny * w + nx;
        if (nx >= 0 && ny >= 0 && nx < w && ny < h && m.data[j] && !labels[j]) { labels[j] = id; stack.push(j); }
      }
    }
    comps.push({ id, area, x: x0, y: y0, w: x1 - x0 + 1, h: y1 - y0 + 1, cx: sx / area, cy: sy / area,
      edge: x0 === 0 || y0 === 0 || x1 === w - 1 || y1 === h - 1 });
  }
  return { labels, comps };
}
```

```ts
import { type RGB, toValue, gaussianBlur } from './image';
import { otsu, binarize } from './threshold';
import { open, close } from './morph';
import { fillHoles } from './fill';
import { components } from './ccl';

export type Calibration = { pxPerMm: number; confidence: number; method: 'fft' | 'manual' };
export type Measurement = { id: number; part: 'larva' | 'head' | 'thorax' | 'abdomen' | 'body'; areaPx: number;
  areaMm2: number; sdMm2?: number; bbox: [number, number, number, number]; flag?: string };
export type LarvaOptions = { minAreaMm2: number; minAspect: number; minFill: number; minThreshold: number };
const DEFAULTS: LarvaOptions = { minAreaMm2: 0.05, minAspect: 1.3, minFill: 0.25, minThreshold: 60 };

export function measureLarvae(img: RGB, cal: Calibration, opt: Partial<LarvaOptions> = {}) {
  const o = { ...DEFAULTS, ...opt }, v = gaussianBlur(toValue(img));
  // Larvae are bright on dark. Otsu adapts to exposure; minThreshold stops it chasing background noise.
  const r = Math.max(1, Math.round(img.w / 1400));    // ≈ the old 5/9 px kernels at 3584 px wide
  const m = fillHoles(close(open(binarize(v, Math.max(otsu(v), o.minThreshold)), r), 2 * r));
  const { labels, comps } = components(m), px2 = cal.pxPerMm ** 2;
  const kept = comps.filter((c) => {
    const aspect = Math.max(c.w, c.h) / Math.max(1, Math.min(c.w, c.h));    // orientation-free, unlike the old h/w
    return c.area / px2 >= o.minAreaMm2 && aspect >= o.minAspect && c.area / (c.w * c.h) >= o.minFill;
  });
  const sorted = kept.map((c) => c.area).sort((a, b) => a - b), median = sorted[Math.floor(sorted.length / 2)] ?? 0;
  const keep = new Set(kept.map((c) => c.id));
  for (let i = 0; i < labels.length; i++) if (!keep.has(labels[i])) labels[i] = 0;
  const measurements: Measurement[] = kept.map((c, k) => {
    const flags = [c.edge && 'edge', kept.length >= 3 && c.area > 1.8 * median && 'merged?'].filter(Boolean);
    return { id: k + 1, part: 'larva', areaPx: c.area, areaMm2: c.area / px2, bbox: [c.x, c.y, c.w, c.h],
      ...(flags.length ? { flag: flags.join(',') } : {}) };
  });
  return { measurements, labels };
}
```

`src/fft.ts`
```ts
// Iterative radix-2 Cooley–Tukey. Returns magnitudes of the non-negative frequencies.
export function fftMag(signal: Float64Array): Float64Array {
  let n = 1; while (n < signal.length) n <<= 1;
  const re = new Float64Array(n), im = new Float64Array(n); re.set(signal);
  for (let i = 1, j = 0; i < n; i++) {                           // bit-reversal permutation
    let bit = n >> 1; for (; j & bit; bit >>= 1) j ^= bit; j ^= bit;
    if (i < j) { [re[i], re[j]] = [re[j], re[i]]; [im[i], im[j]] = [im[j], im[i]]; }
  }
  for (let len = 2; len <= n; len <<= 1) {
    const ang = (-2 * Math.PI) / len, wr = Math.cos(ang), wi = Math.sin(ang);
    for (let i = 0; i < n; i += len) {
      let cr = 1, ci = 0;
      for (let k = 0; k < len / 2; k++) {
        const a = i + k, b = a + len / 2, tr = re[b] * cr - im[b] * ci, ti = re[b] * ci + im[b] * cr;
        re[b] = re[a] - tr; im[b] = im[a] - ti; re[a] += tr; im[a] += ti;
        [cr, ci] = [cr * wr - ci * wi, cr * wi + ci * wr];
      }
    }
  }
  const out = new Float64Array(n / 2 + 1); for (let k = 0; k <= n / 2; k++) out[k] = Math.hypot(re[k], im[k]);
  return out;
}
```

`src/calibrate.ts`
```ts
import { type RGB, toValue } from './image';
import { fftMag } from './fft';
import type { Calibration } from './larva';
export const MIN_CONFIDENCE = 5;

// 1. The ruler's ticks are vertical lines → strongest horizontal gradient energy picks the row band.
// 2. Average that band into a 1-D profile, remove the slow trend, window it, FFT it.
// 3. The biggest peak (periods 3–40 px) is the tick spacing; parabolic interpolation gives sub-bin accuracy.
export function calibrateFromScale(img: RGB, ticksPerMm = 100) {
  const g = toValue(img), { w, h } = g, energy = new Float64Array(h);
  for (let y = 0; y < h; y++) for (let x = 1; x < w; x++) energy[y] += Math.abs(g.data[y * w + x] - g.data[y * w + x - 1]);
  let row = 0, best = -1;
  for (let y = 7; y < h - 7; y++) { let s = 0; for (let k = -7; k <= 7; k++) s += energy[y + k]; if (s > best) { best = s; row = y; } }
  const prof = new Float64Array(w);
  for (let x = 0; x < w; x++) { let s = 0, c = 0; for (let y = Math.max(0, row - 10); y < Math.min(h, row + 10); y++) { s += g.data[y * w + x]; c++; } prof[x] = s / c; }
  const detr = new Float64Array(w);
  for (let x = 0; x < w; x++) { let s = 0, c = 0; for (let k = Math.max(0, x - 25); k <= Math.min(w - 1, x + 25); k++) { s += prof[k]; c++; }
    detr[x] = (prof[x] - s / c) * (0.5 - 0.5 * Math.cos((2 * Math.PI * x) / (w - 1))); }   // Hann window
  const mag = fftMag(detr), n = (mag.length - 1) * 2;
  const lo = Math.ceil(n / 40), hi = Math.floor(n / 3);
  let k = lo; for (let i = lo; i <= hi; i++) if (mag[i] > mag[k]) k = i;
  const a = mag[k - 1], b = mag[k], c = mag[k + 1], kk = k + (0.5 * (a - c)) / (a - 2 * b + c || 1);
  const band = Array.from(mag.slice(lo, hi + 1)).sort((p, q) => p - q), median = band[band.length >> 1] || 1;
  const periodPx = n / kk;
  return { pxPerMm: periodPx * ticksPerMm, confidence: b / median, method: 'fft' as const, row, periodPx };
}

export function manualCalibration(a: [number, number], b: [number, number], mm = 1): Calibration {
  return { pxPerMm: Math.hypot(b[0] - a[0], b[1] - a[1]) / mm, confidence: Infinity, method: 'manual' };
}
```

```ts
import { type RGB, resizeBilinear, gray } from './image';
import { components } from './ccl';
import { open } from './morph';
import type { Calibration, Measurement } from './larva';

export const S = 512, DROP_C = 576, PARTS = ['head', 'thorax', 'abdomen'] as const;
const MEAN = [0.485, 0.456, 0.406], STD = [0.229, 0.224, 0.225];
export type Letterbox = { scale: number; padX: number; padY: number; w: number; h: number };
export type RunModel = (image: Float32Array, drop: Float32Array, T: number) => Promise<Float32Array>;

export function preprocess(img: RGB) {
  const scale = S / Math.max(img.w, img.h), w = Math.round(img.w * scale), h = Math.round(img.h * scale);
  const padX = (S - w) >> 1, padY = (S - h) >> 1, r = resizeBilinear(img, w, h), tensor = new Float32Array(3 * S * S);
  for (let c = 0; c < 3; c++) tensor.fill(-MEAN[c] / STD[c], c * S * S, (c + 1) * S * S);   // pad = black
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) for (let c = 0; c < 3; c++)
    tensor[c * S * S + (y + padY) * S + x + padX] = (r.data[(y * w + x) * 4 + c] / 255 - MEAN[c]) / STD[c];
  return { tensor, box: { scale, padX, padY, w, h } };
}

export function dropMasks(T: number, p = 0.2, rand = Math.random) {
  return Float32Array.from({ length: T * DROP_C }, () => (rand() < p ? 0 : 1 / (1 - p)));
}

export function postprocess(probs: Float32Array, T: number, box: Letterbox, cal: Calibration) {
  const N = S * S, labels = new Uint8Array(N), fg = gray(S, S);
  for (let i = 0; i < N; i++) {                              // argmax of the mean over passes
    let best = 0, bv = -1;
    for (let k = 0; k < 4; k++) { let s = 0; for (let t = 0; t < T; t++) s += probs[(t * 4 + k) * N + i]; if (s > bv) { bv = s; best = k; } }
    labels[i] = best; fg.data[i] = best ? 1 : 0;
  }
  const { labels: inst, comps } = components(open(fg, 2));
  const flies = comps.filter((c) => c.area > 0.002 * N);    // drop crumbs
  const toPx = 1 / box.scale ** 2, px2 = cal.pxPerMm ** 2, out: Measurement[] = [];
  flies.forEach((c, f) => {
    // per pass: count each part inside this fly's region
    const counts = Array.from({ length: T }, () => [0, 0, 0, 0]);
    for (let y = c.y; y < c.y + c.h; y++) for (let x = c.x; x < c.x + c.w; x++) {
      const i = y * S + x; if (inst[i] !== c.id) continue;
      for (let t = 0; t < T; t++) {
        let best = 0, bv = -1; for (let k = 0; k < 4; k++) { const v = probs[(t * 4 + k) * N + i]; if (v > bv) { bv = v; best = k; } }
        counts[t][best]++;
      }
    }
    const stat = (xs: number[]) => { const m = xs.reduce((a, b) => a + b, 0) / xs.length;
      return [m, Math.sqrt(xs.reduce((a, b) => a + (b - m) ** 2, 0) / xs.length)]; };
    const bbox: Measurement['bbox'] = [(c.x - box.padX) / box.scale, (c.y - box.padY) / box.scale, c.w / box.scale, c.h / box.scale];
    const series = [...PARTS.map((_, k) => counts.map((n) => n[k + 1])), counts.map((n) => n[1] + n[2] + n[3])];
    (['head', 'thorax', 'abdomen', 'body'] as const).forEach((part, k) => {
      const [m, sd] = stat(series[k]); if (!m) return;
      out.push({ id: f + 1, part, areaPx: Math.round(m * toPx), areaMm2: (m * toPx) / px2, sdMm2: (sd * toPx) / px2, bbox,
        ...(part === 'body' && sd / m > 0.1 ? { flag: 'uncertain' } : {}), ...(c.edge ? { flag: 'edge' } : {}) });
    });
  });
  return { measurements: out, labels };
}

export async function measureAdult(img: RGB, cal: Calibration, run: RunModel, T = 8) {
  const { tensor, box } = preprocess(img), image = new Float32Array(T * tensor.length);
  for (let t = 0; t < T; t++) image.set(tensor, t * tensor.length);
  const probs = await run(image, dropMasks(T), T);
  return { ...postprocess(probs, T, box, cal), box };
}
```
